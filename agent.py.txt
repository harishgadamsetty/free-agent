import os
import traceback

print("--- DIAGNOSTIC START ---")

# 1. Test importing groq
try:
  from groq import Groq

  print("SUCCESS: Groq library imported correctly.")
except Exception as e:
  print("CRITICAL ERROR importing Groq:")
  traceback.print_exc()
  exit(0)

# 2. Test API Key check
api_key = os.environ.get("GROQ_API_KEY")
if not api_key:
  print(
      "CRITICAL ERROR: GROQ_API_KEY is missing or empty! Check your GitHub"
      " Secrets name."
  )
  exit(0)
else:
  print("SUCCESS: GROQ_API_KEY found.")

# 3. Test API Call
try:
  client = Groq(api_key=api_key)
  print("Sending test request to Groq...")

  completion = client.chat.completions.create(
      model="llama-3.3-70b-versatile",
      messages=[
          {
              "role": "system",
              "content": "You are a helpful 24/7 autonomous cloud assistant.",
          },
          {"role": "user", "content": "Say 'Agent is fully operational!'"},
      ],
      max_tokens=100,
  )

  print("--- AGENT EXECUTION SUCCESS ---")
  print(completion.choices[0].message.content)

except Exception as e:
  print("CRITICAL ERROR during API call:")
  traceback.print_exc()

print("--- DIAGNOSTIC END ---")
exit(0)
