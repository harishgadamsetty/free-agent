import os
import time
from groq import Groq

# Check if the API key is loaded properly
api_key = os.environ.get("GROQ_API_KEY")
if not api_key:
  print(
      "ERROR: GROQ_API_KEY environment variable is missing! Check your GitHub"
      " Secrets."
  )
  exit(1)

# Initialize the Groq client
client = Groq(api_key=api_key)


def run_agent_task():
  try:
    print("Sending request to Groq AI...")
    completion = client.chat.completions.create(
        model="llama-3.3-70b-versatile",
        messages=[
            {
                "role": "system",
                "content": (
                    "You are a helpful 24/7 autonomous cloud assistant."
                ),
            },
            {
                "role": "user",
                "content": (
                    "Perform your scheduled background check and report status."
                ),
            },
        ],
        temperature=0.7,
        max_tokens=300,
    )

    print("--- Agent Execution Success ---")
    print(completion.choices[0].message.content)

  except Exception as e:
    print(f"An error occurred during API execution: {e}")
    exit(1)


if __name__ == "__main__":
  print("Agent started successfully in the cloud.")
  run_agent_task()
